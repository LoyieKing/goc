# FALLBACK only (P11): prefer pass/harness.mir → goobj/mirparse.
# Full per-function MI lists for goobj (physreg/SP/symbol). Packaged from
# post-pass MIR + ABIInternal/goobj contracts. Lowered by goobj/mirlower.
#
# P10 NOSPLIT vs checked policy:
# - flags nosplit + MIR morestack (MOV FS:-8 / CMP / JCC / CALL morestack / JMP):
#   MIR owns the stack check; Go assembler must NOT insert a dual-track auto prologue.
# - flags nosplit without morestack: tiny leaf helpers that cannot grow (goc_leaf).
# - Never: non-NOSPLIT goc TEXT relying on Go auto stack-check prologue (removed P10).

.begin_fn goc_checked_add
go_sym main.GocCheckedAdd
frame 0
flags nosplit
.label entry
op=MOV64rm dst=R11 mem=FS:-8
op=CMP64rm reg=RSP mem=16(R11)
op=JCC_1 cond=HI target=ok
op=ADD64mi8 mem=main.morestackHits imm=1
op=CALL64pcrel32 sym=runtime.morestack_noctxt
op=JMP_1 target=entry
.label ok
op=MOV64rr dst=DI src=AX
op=MOV64rr dst=SI src=BX
op=CALL64pcrel32 sym=goc_leaf
op=RET64
.end_fn

op=JMP_1 target=entry

op=MOV64rm dst=AX mem=(AX)
op=RET64
.end_fn
